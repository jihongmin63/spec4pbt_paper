\section{Evaluation}
\label{sec:evaluation}

\subsection{RQ1: Effectiveness of Predicate Reformulation}

\emph{How does iterative verified predicate reformulation affect generation
efficiency and input diversity compared with existing approaches?}

\subsection{RQ2: Efficiency, Cost, and Consistency Across Agent Runs}

\emph{How does the approach compare with direct agentic generator construction
and verification in generation efficiency, construction cost, and their
variability across independent runs?}

\subsection{RQ3: Applicability to Complex Predicates}

\emph{How effectively can the approach improve generation efficiency for complex
predicates from real-world language specifications?}

\subsection{RQ4: Bug Finding in Language Implementations}

\emph{Can the derived generators expose bugs in real-world language
implementations?}

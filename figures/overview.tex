% Figure 1: 명세에서 작성한 SpecTec-like 성질의 세 산출물.
% 파랑은 Lean 산출물, 보라는 외부 기계적 절차, 주황은 에이전트가 쓰는 증명이다.
% 2번과 3번은 아래 implementation PBT에 사용된다.
\begin{tikzpicture}[
  font=\footnotesize,
  box/.style={draw, rounded corners=2pt, align=center, inner sep=5pt},
  input/.style={box, very thick, fill=white},
  artifact/.style={
    chamfered rectangle, chamfered rectangle corners=north east,
    chamfered rectangle xsep=1.8mm, draw=blue!60!black, thick,
    fill=blue!7, align=center, text width=3.55cm,
    minimum height=2.1cm, inner sep=5pt},
  external/.style={artifact, draw=purple!65!black, fill=purple!9},
  edge/.style={-{Stealth[length=1.8mm]}, thick},
  derive/.style={edge, draw=black!60},
  leantranslate/.style={derive, draw=blue!60!black},
  externaltranslate/.style={derive, draw=purple!65!black},
  agent/.style={edge, very thick, draw=orange!75!black},
  flow/.style={edge, very thick, draw=blue!60!black},
  externalflow/.style={flow, draw=purple!65!black},
  lbl/.style={font=\scriptsize, align=center, inner sep=2pt},
]
  % A property is written over the specification before any artifacts are derived.
  \node[input, text width=2.6cm, minimum height=0.9cm]
    (spec) at (0, 0) {SpecTec\\specification};
  \node[box, draw=orange!75!black, thick, fill=orange!9,
        text width=3.55cm, minimum height=0.9cm]
    (property) at (4.6, 0) {\textbf{SpecTec-like property}};
  \draw[agent] (spec) -- node[lbl, above, text=orange!75!black]
    {agent\\writes} (property);

  % Three artifacts derived from the same property.
  \node[artifact] (lean) at (0, -2.75) {
    \textbf{1. Property in Lean}\\[6pt]
    {\scriptsize\color{orange!75!black} Agent-written Lean proof:}\\
    {\scriptsize property validity}};
  \node[artifact] (generator) at (4.6, -2.75) {
    \textbf{2. Input generator}\\
    {\scriptsize Lean code for premise inputs}\\[6pt]
    {\scriptsize\color{orange!75!black} Agent-written Lean proof:}\\
    {\scriptsize soundness and completeness}};
  \node[external] (oracle) at (9.2, -2.75) {
    \textbf{3. Result checker}\\
    {\scriptsize external mechanical procedure}\\
    {\scriptsize decides the conclusion}\\
    {\scriptsize on observed results}};

  \coordinate (branch) at (4.6, -1.1);
  \draw[thick, draw=black!60] (property.south) -- (branch);
  \draw[leantranslate] (branch) -| node[lbl, above, pos=0.25,
    text=blue!60!black]
    {translate to Lean} (lean.north);
  \draw[leantranslate] (branch) -- node[lbl, right, text=blue!60!black]
    {to Lean} (generator.north);
  \draw[externaltranslate] (branch) -| node[lbl, above, pos=0.25,
    text=purple!65!black]
    {translate externally} (oracle.north);

  % The generator supplies inputs; the checker judges implementation results.
  \node[input, text width=2.65cm, minimum height=0.9cm]
    (implementation) at (4.6, -5.3) {Run\\implementation};
  \node[box, draw=purple!65!black, thick, fill=white,
        text width=2.65cm, minimum height=0.9cm]
    (verdict) at (9.2, -5.3) {Apply result checker\\{\scriptsize pass / fail}};

  \begin{scope}[on background layer]
    \node[draw=black!35, fill=black!3, rounded corners=3pt,
          fit=(implementation) (verdict), inner sep=9pt] (pbt) {};
  \end{scope}
  \node[anchor=north west, font=\footnotesize\bfseries, inner sep=0pt]
    at ([yshift=-4pt]pbt.south west) {Implementation PBT};

  \draw[flow] (generator.south) -- node[lbl, left]
    {generate inputs} (implementation.north);
  \draw[externalflow] (oracle.south) -- node[lbl, right]
    {use checker} (verdict.north);
  \draw[edge] (implementation.east) -- node[lbl, above]
    {observed\\results} (verdict.west);
\end{tikzpicture}

structure Input where
  p : p4program

def Premise (input : Input) : Prop :=
  ¬ TypeChecks input.p

def Goal (input : Input) : Prop :=
  Rejected input.p

def property : Prop :=
  ∀ input : Input,
    Premise input → Goal input
